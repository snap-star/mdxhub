### 11.1 Destructuring

```javascript
// === Array Destructuring ===
const [first, second] = ["apple", "banana", "cherry"];
console.log(first);   // "apple"
console.log(second);  // "banana"

// Skip elements
const [one, , three] = [1, 2, 3];
console.log(one, three);  // 1 3

// Rest in destructuring
const [head, ...tail] = [1, 2, 3, 4];
console.log(head);  // 1
console.log(tail);  // [2, 3, 4]

// Default values
const [a = 10, b = 20] = [5];
console.log(a, b);  // 5 20

// Swap variables
let x = 1, y = 2;
[x, y] = [y, x];
console.log(x, y);  // 2 1

// === Object Destructuring ===
const user = { name: "Alice", age: 30, city: "NYC" };

const { name, age } = user;
console.log(name, age);  // "Alice" 30

// Rename variables
const { name: userName, age: userAge } = user;
console.log(userName);  // "Alice"

// Default values
const { role = "user" } = user;
console.log(role);  // "user" (not in object, so default used)

// Nested destructuring
const { address: { city, zip = "00000" } } = { address: { city: "NYC" } };
console.log(city, zip);  // "NYC" "00000"

// Rest properties (ES2018)
const { name, ...rest } = { name: "Alice", age: 30, city: "NYC" };
console.log(rest);  // { age: 30, city: "NYC" }

// === Function Parameter Destructuring ===

// Instead of:
function displayUser(user) {
  console.log(`${user.name} is ${user.age}`);
}

// Write:
function displayUser({ name, age }) {
  console.log(`${name} is ${age}`);
}
displayUser({ name: "Alice", age: 30 });  // "Alice is 30"

// With defaults
function createButton({ text = "Click", color = "blue", onClick } = {}) {
  return { text, color, onClick };
}
createButton();  // { text: "Click", color: "blue", onClick: undefined }
createButton({ text: "Submit", color: "green" });

// Nested destructuring in parameters
function drawChart({ data, config: { width = 400, height = 300 } }) {
  console.log(`Drawing ${data.length} points in ${width}x${height}`);
}
drawChart({ data: [1, 2, 3], config: { width: 800 } });
```

### 11.2 Spread Syntax

```javascript
// === Array Spread ===
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];

// Combine arrays
const combined = [...arr1, ...arr2];           // [1, 2, 3, 4, 5, 6]
const withExtra = [0, ...arr1, 3.5, ...arr2];  // [0, 1, 2, 3, 3.5, 4, 5, 6]

// Copy array (shallow)
const copy = [...arr1];        // [1, 2, 3]

// Convert iterable to array
const fromSet = [...new Set([1, 1, 2, 3])];  // [1, 2, 3]
const fromString = [..."hello"];               // ["h", "e", "l", "l", "o"]

// === Object Spread ===
const defaults = { theme: "light", fontSize: 14 };
const overrides = { fontSize: 16 };

const settings = { ...defaults, ...overrides };  // { theme: "light", fontSize: 16 }

// Add/modify properties
const user = { name: "Alice" };
const withAge = { ...user, age: 30 };             // { name: "Alice", age: 30 }
const renamed = { ...user, name: "Bob" };          // { name: "Bob" }

// Shallow copy warning
const nested = { data: { value: 1 } };
const copied = { ...nested };
copied.data.value = 2;
console.log(nested.data.value);  // 2! (nested object is shared reference)

// === Function Arguments ===
function sum(...numbers) {
  return numbers.reduce((a, b) => a + b, 0);
}
const nums = [1, 2, 3, 4];
sum(...nums);  // Equivalent to sum(1, 2, 3, 4), returns 10

// Math.max with array
Math.max(...[10, 5, 20, 3]);  // 20
```

### 11.4 Optional Chaining (`?.`)

```javascript
const user = {
  profile: {
    address: {
      city: "NYC"
    }
  }
};

// Without optional chaining (verbose)
const city1 = user && user.profile && user.profile.address && user.profile.address.city;

// With optional chaining (ES2020)
const city2 = user?.profile?.address?.city;        // "NYC"
const zip2 = user?.profile?.address?.zip;           // undefined (no error!)
const deep = user?.contacts?.friends?.[0]?.name;    // undefined (safely handles each missing level)

// With function calls
const result = someObject?.maybeAMethod?.();         // Calls if exists, undefined if not

// With bracket notation
const key = "dynamicKey";
const value = user?.[key]?.nested;

// Cannot chain after optional chain without another `?.`
// user?.profile.address.city  // ❌ If profile is null, .address throws!
// user?.profile?.address?.city  // ✓ Safe at every level
```

### 11.5 Nullish Coalescing (`??`)

```javascript
// Returns right operand ONLY if left is null or undefined
const value1 = null ?? "default";       // "default"
const value2 = undefined ?? "default";  // "default"
const value3 = 0 ?? "default";          // 0 (0 is not nullish!)
const value4 = "" ?? "default";         // "" (empty string is not nullish!)
const value5 = false ?? "default";      // false (false is not nullish!)

// Real-world: provide defaults for potentially missing config
const config = {
  timeout: 0,       // 0 means "no timeout" — valid value!
  retries: null,    // null means "use default"
};

const timeout = config.timeout ?? 5000;    // 0 ✓ (0 is kept!)
const retries = config.retries ?? 3;        // 3 (null is nullish)

// Comparison: || vs ??
const port = 0;
console.log(port || 3000);   // 3000 (0 is falsy — WRONG for port numbers!)
console.log(port ?? 3000);   // 0 (0 is not nullish — CORRECT!)

// Combining with optional chaining
const userCity = user?.profile?.address?.city ?? "Unknown";
```

### 12.2 Promises

A **Promise** represents a value that may not exist yet but will be resolved at some point.

```javascript
// Creating a Promise
const promise = new Promise((resolve, reject) => {
  // Async operation
  setTimeout(() => {
    const success = true;
    if (success) {
      resolve("Operation completed!");  // Fulfill the promise
    } else {
      reject(new Error("Operation failed!"));  // Reject the promise
    }
  }, 1000);
});

// Consuming a Promise
promise
  .then(result => {
    console.log(result);  // "Operation completed!"
    return result.toUpperCase();
  })
  .then(upper => {
    console.log(upper);   // "OPERATION COMPLETED!"
  })
  .catch(error => {
    console.error(error.message);
  })
  .finally(() => {
    console.log("Cleanup: always runs");
  });

// Promise states:
// 1. PENDING — initial state, neither fulfilled nor rejected
// 2. FULFILLED — operation completed successfully
// 3. REJECTED — operation failed
// Once settled (fulfilled or rejected), a promise never changes state.

// Promisifying callbacks
function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

delay(1000).then(() => console.log("1 second passed"));

// Real-world: fetch API (returns a Promise)
fetch("https://api.example.com/users")
  .then(response => {
    if (!response.ok) throw new Error(`HTTP ${response.status}`);
    return response.json();  // Returns another Promise
  })
  .then(data => console.log(data))
  .catch(error => console.error("Fetch error:", error));
```

### 12.3 Promise Static Methods

```javascript
// Promise.resolve — create a resolved promise
Promise.resolve(42).then(v => console.log(v));  // 42

// Promise.reject — create a rejected promise
Promise.reject(new Error("Oops")).catch(e => console.error(e));

// Promise.all — wait for ALL promises (fail fast)
const urls = ["/api/users", "/api/posts", "/api/comments"];
Promise.all(urls.map(url => fetch(url).then(r => r.json())))
  .then(([users, posts, comments]) => {
    console.log(`Loaded ${users.length} users, ${posts.length} posts`);
  })
  .catch(error => console.error("One request failed:", error));

// Promise.allSettled — wait for ALL, never fails
Promise.allSettled([
  Promise.resolve("success"),
  Promise.reject("error"),
  Promise.resolve("another success")
]).then(results => {
  console.log(results);
  // [
  //   { status: "fulfilled", value: "success" },
  //   { status: "rejected", reason: "error" },
  //   { status: "fulfilled", value: "another success" }
  // ]
});

// Promise.race — first to settle wins
Promise.race([
  fetch("/api/data"),
  new Promise((_, reject) => 
    setTimeout(() => reject(new Error("Timeout!")), 5000)
  )
]).then(response => console.log("Got response before timeout!"));

// Promise.any — first to FULFILL wins (ignores rejections unless all reject)
Promise.any([
  fetch("/api/primary"),
  fetch("/api/backup"),
  fetch("/api/fallback")
]).then(response => console.log("At least one succeeded!"));

// ES2024: Grouped async operations
const promises = [
  fetch("/api/a"),
  fetch("/api/b"),
  fetch("/api/c"),
];

// Wait for all, but get results individually with their status
const results = await Promise.allSettled(promises);
const succeeded = results
  .filter(r => r.status === "fulfilled")
  .map(r => r.value);
```

### 12.4 Async/Await

`async/await` is **syntactic sugar over Promises** — it makes async code look synchronous.

```javascript
// Mark function as async
async function fetchUser(id) {
  const response = await fetch(`/api/users/${id}`);
  if (!response.ok) throw new Error(`HTTP ${response.status}`);
  return await response.json();
}

// Using async function
async function init() {
  try {
    const user = await fetchUser(1);
    console.log(user.name);
  } catch (error) {
    console.error("Failed:", error.message);
  }
}

// Async arrow function
const fetchPosts = async (userId) => {
  const res = await fetch(`/api/users/${userId}/posts`);
  return res.json();
};

// An async function ALWAYS returns a Promise
async function getValue() {
  return 42;  // Automatically wrapped in Promise.resolve(42)
}
getValue().then(v => console.log(v));  // 42

// Sequential execution (one after another)
async function sequential() {
  const user = await fetchUser(1);       // Wait for user
  const posts = await fetchPosts(user.id); // Then fetch posts
  const comments = await fetchComments(posts[0].id); // Then comments
  return comments;
}
// Total time: user_time + posts_time + comments_time

// Parallel execution (all at once)
async function parallel() {
  const [users, posts, comments] = await Promise.all([
    fetch("/api/users").then(r => r.json()),
    fetch("/api/posts").then(r => r.json()),
    fetch("/api/comments").then(r => r.json()),
  ]);
  return { users, posts, comments };
}
// Total time: max(user_time, posts_time, comments_time)

// Mixed: start parallel, then sequential
async function mixed() {
  // Start both at the same time
  const userPromise = fetchUser(1);
  const settingsPromise = fetchSettings(1);
  
  // Wait for both
  const [user, settings] = await Promise.all([userPromise, settingsPromise]);
  
  // Then do something that needs both
  const personalized = await personalize(user, settings);
  return personalized;
}

// Top-level await (in modules)
// data.js
const data = await fetch("/api/config").then(r => r.json());
export default data;

// Looping with async/await
async function processUsers(userIds) {
  const results = [];
  for (const id of userIds) {
    // Sequential: one at a time (polite to the server)
    const user = await fetchUser(id);
    results.push(user);
  }
  return results;
}

// ⚠️ forEach doesn't work with async!
// WRONG: All fetches start simultaneously, results lost
userIds.forEach(async (id) => {
  const user = await fetchUser(id);  // Fire-and-forget!
});

// CORRECT: Use for...of for sequential
// CORRECT: Use Promise.all + map for parallel
const users = await Promise.all(userIds.map(id => fetchUser(id)));
```

### 13.1 Class Basics

```javascript
class Person {
  // Static property (shared across all instances)
  static species = "Homo sapiens";
  static count = 0;

  // Public field
  name;
  
  // Private field (only accessible within class)
  #age;
  
  // Private static field
  static #idCounter = 0;

  constructor(name, age) {
    this.name = name;
    this.#age = age;
    Person.count++;
    this.#id = ++Person.#idCounter;
  }

  // Getter
  get age() {
    return this.#age;
  }

  // Setter with validation
  set age(value) {
    if (value < 0 || value > 150) {
      throw new Error("Invalid age");
    }
    this.#age = value;
  }

  // Private method
  #validate() {
    return this.name && this.#age > 0;
  }

  // Public method
  greet() {
    if (!this.#validate()) return "Invalid person";
    return `Hello, I'm ${this.name} and I'm ${this.#age} years old.`;
  }

  // Static method
  static createAnonymous() {
    return new Person("Anonymous", 0);
  }

  // Static factory
  static fromJSON(json) {
    const data = JSON.parse(json);
    return new Person(data.name, data.age);
  }
}

// Usage
const alice = new Person("Alice", 30);
console.log(alice.name);       // "Alice"
console.log(alice.age);        // 30 (uses getter)
alice.age = 31;                // Uses setter
console.log(alice.greet());    // "Hello, I'm Alice..."
console.log(Person.species);   // "Homo sapiens"
console.log(Person.count);     // 1

// Private fields are truly private
// console.log(alice.#age);    // ❌ SyntaxError!

// Static methods
const anon = Person.createAnonymous();
const fromData = Person.fromJSON('{"name":"Bob","age":25}');
```

### 13.2 Inheritance

```javascript
class Employee extends Person {
  // Public static
  static roles = ["developer", "designer", "manager"];

  #salary;
  #role;

  constructor(name, age, role, salary) {
    super(name, age);           // Must call super() before using 'this'
    this.#role = role;
    this.#salary = salary;
  }

  get salary() {
    return this.#salary;
  }

  // Override parent method
  greet() {
    const base = super.greet();  // Call parent method
    return `${base} I work as a ${this.#role}.`;
  }

  // New method
  giveRaise(percent) {
    this.#salary *= (1 + percent / 100);
    return this.#salary;
  }

  // Check instance type
  static isEmployee(obj) {
    return obj instanceof Employee;
  }
}

const dev = new Employee("Alice", 30, "developer", 100000);
console.log(dev.greet());
// "Hello, I'm Alice and I'm 30 years old. I work as a developer."
console.log(dev instanceof Employee);  // true
console.log(dev instanceof Person);    // true
console.log(Employee.isEmployee(dev)); // true
```

```javascript
class Example {
```

### 14.1 ES Modules (ESM)

```javascript
// === Exporting ===

// math.js

// Named exports
export const PI = 3.14159;
export const E = 2.71828;

export function add(a, b) {
  return a + b;
}

export function multiply(a, b) {
  return a * b;
}

const utils = { log, warn, error };
export { utils };

// Default export (one per module)
export default class Calculator {
  // ...
}

// Re-export (barrel file pattern)
export { add, subtract } from "./math.js";
export { default as Calculator } from "./calculator.js";
export * from "./helpers.js";

// === Importing ===

// app.js

// Named imports
import { PI, add } from "./math.js";

// Import with alias
import { multiply as mult } from "./math.js";

// Import everything as namespace
import * as math from "./math.js";
console.log(math.PI);       // 3.14159
console.log(math.add(2, 3)); // 5

// Default import
import Calculator from "./calculator.js";

// Default + named imports
import Calculator, { PI, add } from "./math.js";

// Import for side effects (runs module code, no imports)
import "./analytics.js";

// Dynamic import (returns a Promise)
async function loadHeavyModule() {
  const { heavyFunction } = await import("./heavy.js");
```

### 15.1 try/catch/finally

```javascript
// Basic error handling
try {
  const data = JSON.parse(userInput);
  processData(data);
} catch (error) {
  console.error("Parsing failed:", error.message);
  showUserMessage("Invalid input format");
} finally {
  // Always runs (cleanup)
  hideLoadingSpinner();
}

// Catch specific error types
try {
  riskyOperation();
} catch (error) {
  if (error instanceof SyntaxError) {
    console.error("Syntax issue:", error.message);
  } else if (error instanceof TypeError) {
    console.error("Type issue:", error.message);
  } else if (error instanceof NetworkError) {
    console.error("Network issue — retrying...");
    retry();
  } else {
    throw error;  // Re-throw unknown errors
  }
}

// Optional catch binding (ES2019)
try {
  riskyOperation();
} catch {
  // Error object not needed
  console.error("Something went wrong");
}

// Throwing custom errors
function divide(a, b) {
  if (b === 0) throw new Error("Cannot divide by zero");
  if (typeof a !== "number" || typeof b !== "number") {
    throw new TypeError("Arguments must be numbers");
  }
  return a / b;
}

// Custom error classes
class ValidationError extends Error {
  constructor(field, message) {
    super(message);
    this.name = "ValidationError";
    this.field = field;
  }
}

class NetworkError extends Error {
  constructor(url, status) {
    super(`Request to ${url} failed with status ${status}`);
    this.name = "NetworkError";
    this.status = status;
    this.url = url;
  }
}

// Usage
try {
  validateUser({ email: "invalid" });
} catch (error) {
  if (error instanceof ValidationError) {
    console.error(`${error.field}: ${error.message}`);
  }
}
```

### 15.3 Defensive Programming Patterns

```javascript
// Guard clauses — fail fast
function processUser(user) {
  if (!user) throw new ValidationError("user", "Required");
  if (!user.email) throw new ValidationError("email", "Required");
  if (!isValidEmail(user.email)) throw new ValidationError("email", "Invalid format");
  if (!user.age || user.age < 0) throw new ValidationError("age", "Must be positive");
  
  // Main logic (only reached if all guards pass)
  return saveToDatabase(user);
}

// Assertion function
function assert(condition, message) {
  if (!condition) throw new Error(`Assertion failed: ${message}`);
}

function calculateDiscount(price, rate) {
  assert(typeof price === "number" && price >= 0, "Price must be non-negative");
  assert(typeof rate === "number" && rate >= 0 && rate <= 1, "Rate must be 0-1");
  return price * rate;
}

// Result type pattern (avoid throwing for expected failures)
function safeDivide(a, b) {
  if (b === 0) return { ok: false, error: "Cannot divide by zero" };
  return { ok: true, value: a / b };
}

const result = safeDivide(10, 0);
if (result.ok) {
  console.log(result.value);
} else {
  console.error(result.error);
}
```

```javascript
// WeakMap — keys must be objects, weakly held (garbage collectible)
const privateData = new WeakMap();

class SecureUser {
  constructor(name, password) {
    privateData.set(this, { password });
    this.name = name;
  }

  }
}
```

### 18.2 Building a Reusable API Client

```javascript
// api/client.js
class ApiClient {
  constructor(baseURL, defaultOptions = {}) {
    this.baseURL = baseURL.replace(/\/$/, "");  // Remove trailing slash
    this.defaultOptions = defaultOptions;
  }

  async request(endpoint, options = {}) {
    const url = `${this.baseURL}${endpoint}`;
    const config = {
      ...this.defaultOptions,
      ...options,
      headers: {
        "Content-Type": "application/json",
        ...this.defaultOptions.headers,
        ...options.headers,
      },
    };

    if (config.body && typeof config.body === "object") {
      config.body = JSON.stringify(config.body);
    }

    const response = await fetch(url, config);
    return this.handleResponse(response);
  }

  async handleResponse(response) {
    if (!response.ok) {
      const error = await response.json().catch(() => ({}));
      throw new ApiError(response.status, error.message || "Request failed");
    }
    
    if (response.status === 204) return null;  // No content
    return response.json();
  }

  // Convenience methods
  get(endpoint, options) {
    return this.request(endpoint, { ...options, method: "GET" });
  }
  
  post(endpoint, body, options) {
    return this.request(endpoint, { ...options, method: "POST", body });
  }
  
  put(endpoint, body, options) {
    return this.request(endpoint, { ...options, method: "PUT", body });
  }
  
  patch(endpoint, body, options) {
    return this.request(endpoint, { ...options, method: "PATCH", body });
  }
  
  delete(endpoint, options) {
    return this.request(endpoint, { ...options, method: "DELETE" });
  }
}

class ApiError extends Error {
  constructor(status, message) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }
}

// Usage
const api = new ApiClient("https://api.example.com", {
  headers: { "X-API-Version": "v2" },
});

// With auth interceptor
api.request = new Proxy(api.request, {
  apply(target, thisArg, args) {
    const [endpoint, options = {}] = args;
    options.headers = {
      ...options.headers,
      Authorization: `Bearer ${localStorage.getItem("token")}`,
    };
    return target.call(thisArg, endpoint, options);
  },
});

// Use it!
const users = await api.get("/users");
const user = await api.post("/users", { name: "Alice", email: "alice@example.com" });
await api.delete(`/users/${user.id}`);
```
